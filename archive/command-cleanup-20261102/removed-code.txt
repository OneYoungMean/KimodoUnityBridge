## Superseded loop constraint builder

internal static string BuildLoopConstraintJson(
            KimodoRawMotionData motion,
            string modelName,
            int runtimeTrimStartFrame,
            int targetFrameCount,
            int runtimeFrameCount,
            float frameRate,
            bool lockPositionX = false,
            bool lockPositionY = true,
            bool lockPositionZ = false,
            bool lockRotationX = true,
            bool lockRotationY = false,
            bool lockRotationZ = true)
        {
            if (motion == null || motion.FrameCount < 2 || motion.FrameRate <= 0f ||
                runtimeTrimStartFrame < 0 || targetFrameCount <= 1 || runtimeFrameCount <= 0 || frameRate <= 0f)
            {
                throw new InvalidOperationException("Loop constraint frame range is invalid.");
            }

            int anchorSourceFrame = ResolveLoopAnchorFrameForGeneration(motion);
            if (!TryBuildFullBodyFrame(
                    motion, modelName, anchorSourceFrame,
                    out KimodoConstraintInternalData first, out string anchorError))
            {
                throw new InvalidOperationException(
                    $"Loop pass 1 anchor pose sampling failed at frame {anchorSourceFrame}: {anchorError}");
            }
            if (!TryBuildFullBodyFrame(
                    motion, modelName, motion.FrameCount - 1,
                    out KimodoConstraintInternalData tail, out string tailError))
            {
                throw new InvalidOperationException(
                    $"Loop pass 1 tail pose sampling failed at frame {motion.FrameCount - 1}: {tailError}");
            }
            if (!TryEstimatePlanarVelocity(motion, out Vector3 averageVelocity, out string velocityError))
            {
                throw new InvalidOperationException(
                    $"Loop pass 1 average velocity estimation failed: {velocityError}");
            }

            int terminalFrame = runtimeTrimStartFrame + targetFrameCount - 1;
            int virtualTailFrame = runtimeFrameCount - 1;
            if (terminalFrame >= runtimeFrameCount || virtualTailFrame < terminalFrame)
            {
                throw new InvalidOperationException("Loop terminal frame is outside the runtime range.");
            }

            Quaternion firstRotation = ResolveRootRotation(first);
            Quaternion tailRotation = ResolveRootRotation(tail);
            float targetSpanSeconds = targetFrameCount / frameRate;
            Vector3 projectedDelta = averageVelocity * targetSpanSeconds;
            projectedDelta.y = 0f;
            Vector3 projectedTerminalRoot = first.rootPosition + projectedDelta;
            Vector3 terminalRoot = new Vector3(
                projectedTerminalRoot.x,
                tail.rootPosition.y,
                projectedTerminalRoot.z);
            Quaternion terminalHeading = tailRotation;
            ApplyLoopAnchorAxisLocks(
                first,
                ref terminalRoot,
                ref terminalHeading,
                lockPositionX,
                lockPositionY,
                lockPositionZ,
                lockRotationX,
                lockRotationY,
                lockRotationZ);
            Vector3 planarDelta = terminalRoot - first.rootPosition;
            planarDelta.y = 0f;
            float headRatio = runtimeTrimStartFrame / (float)targetFrameCount;
            float tailRatio = (virtualTailFrame - terminalFrame) / (float)targetFrameCount;
            Vector3 virtualHeadPosition = first.rootPosition - planarDelta * headRatio;
            Vector3 virtualTailPosition = terminalRoot + planarDelta * tailRatio;
            virtualHeadPosition.y = first.rootPosition.y;
            virtualTailPosition.y = terminalRoot.y;

            float firstYaw = KimodoMotionMath.ResolvePlanarYawDegrees(firstRotation);
            float tailYaw = KimodoMotionMath.ResolvePlanarYawDegrees(terminalHeading);
            float yawDelta = Mathf.DeltaAngle(firstYaw, tailYaw);
            Quaternion virtualHeadHeading = Quaternion.Euler(0f, firstYaw - yawDelta * headRatio, 0f);
            Quaternion virtualTailHeading = Quaternion.Euler(0f, tailYaw + yawDelta * tailRatio, 0f);

            first.sampleTime = runtimeTrimStartFrame / (double)frameRate;
            KimodoConstraintInternalData terminal = first.Clone();
            terminal.sampleTime = terminalFrame / (double)frameRate;
            terminal.rootPosition = terminalRoot;
            terminal.localJointAxisAngles[0] =
                KimodoConstraintRotationUtility.QuaternionToAxisAngleVector(terminalHeading);

            var constraints = new JArray
            {
                BuildRoot2DConstraint(
                    new[] { 0, virtualTailFrame },
                    new[] { virtualHeadPosition, virtualTailPosition },
                    new[] { virtualHeadHeading, virtualTailHeading }),
                BuildFullBodyConstraints(
                    new[] { first, terminal },
                    frameRate,
                    runtimeFrameCount / (double)frameRate,
                    includeSmoothRoot: true)[0]
            };
            return constraints.ToString(Formatting.None);
        }



## Unused foot-contact selector chain and velocity estimate

private static bool TryEstimatePlanarVelocity(
            KimodoRawMotionData motion,
            out Vector3 velocity,
            out string error)
        {
            velocity = Vector3.zero;
            error = string.Empty;
            if (motion == null || motion.FrameCount < 2 || motion.FrameRate <= 0f)
            {
                error = "Raw motion does not contain enough timed frames.";
                return false;
            }

            double meanFrame = (motion.FrameCount - 1) * 0.5;
            double denominator = 0.0;
            double numeratorX = 0.0;
            double numeratorZ = 0.0;
            for (int frame = 0; frame < motion.FrameCount; frame++)
            {
                if (!motion.TryReadUnityRootPosition(frame, out Vector3 position))
                {
                    error = $"Raw motion root position is invalid at frame {frame}.";
                    return false;
                }

                double centeredFrame = frame - meanFrame;
                denominator += centeredFrame * centeredFrame;
                numeratorX += position.x * centeredFrame;
                numeratorZ += position.z * centeredFrame;
            }

            if (denominator <= 1e-9)
            {
                error = "Raw motion frame times have no span.";
                return false;
            }

            velocity = new Vector3(
                (float)(numeratorX / denominator * motion.FrameRate),
                0f,
                (float)(numeratorZ / denominator * motion.FrameRate));
            return true;
        }

internal static int ResolveLoopAnchorFrameFromFootContactExchange(KimodoRawMotionData motion)
        {
            if (motion == null || !motion.HasFootContacts || motion.FrameCount < 2)
            {
                return 0;
            }

            for (int liftFrame = 1; liftFrame < motion.FrameCount - 1; liftFrame++)
            {
                if (!TryReadFootState(
                        motion,
                        liftFrame,
                        out bool leftFullyLifted,
                        out _,
                        out bool rightFullyLifted,
                        out _))
                {
                    return 0;
                }

                if (!TryReadFootState(
                        motion,
                        liftFrame - 1,
                        out bool previousLeftFullyLifted,
                        out _,
                        out bool previousRightFullyLifted,
                        out _))
                {
                    return 0;
                }

                bool leftLiftedThisFrame = leftFullyLifted && !previousLeftFullyLifted;
                bool rightLiftedThisFrame = rightFullyLifted && !previousRightFullyLifted;
                if (leftLiftedThisFrame && TryFindFollowingLandingFrame(
                        motion,
                        liftFrame,
                        isLeftFoot: false,
                        out int rightLandingFrame))
                {
                    return liftFrame + (rightLandingFrame - liftFrame) / 2;
                }

                if (rightLiftedThisFrame && TryFindFollowingLandingFrame(
                        motion,
                        liftFrame,
                        isLeftFoot: true,
                        out int leftLandingFrame))
                {
                    return liftFrame + (leftLandingFrame - liftFrame) / 2;
                }
            }

            return 0;
        }

        private static bool TryFindFollowingLandingFrame(
            KimodoRawMotionData motion,
            int liftFrame,
            bool isLeftFoot,
            out int landingFrame)
        {
            landingFrame = -1;
            if (!TryReadFootState(
                    motion,
                    liftFrame,
                    out _,
                    out bool initialLeftFullyLanded,
                    out _,
                    out bool initialRightFullyLanded))
            {
                return false;
            }
            bool wasFullyLanded = isLeftFoot
                ? initialLeftFullyLanded
                : initialRightFullyLanded;

            for (int frame = liftFrame + 1; frame < motion.FrameCount; frame++)
            {
                if (!TryReadFootState(
                        motion,
                        frame,
                        out _,
                        out bool leftFullyLanded,
                        out _,
                        out bool rightFullyLanded))
                {
                    return false;
                }

                bool oppositeFootLanded = isLeftFoot ? leftFullyLanded : rightFullyLanded;
                if (oppositeFootLanded && !wasFullyLanded)
                {
                    landingFrame = frame;
                    return true;
                }
                wasFullyLanded = oppositeFootLanded;
            }

            return false;
        }

        private static bool TryReadFootState(
            KimodoRawMotionData motion,
            int frame,
            out bool leftFullyLifted,
            out bool leftFullyLanded,
            out bool rightFullyLifted,
            out bool rightFullyLanded)
        {
            leftFullyLifted = leftFullyLanded = rightFullyLifted = rightFullyLanded = false;
            if (motion == null || !motion.HasFootContacts || frame < 0 || frame >= motion.FrameCount ||
                !motion.TryReadFootContact(frame, 0, out float leftHeel) ||
                !motion.TryReadFootContact(frame, 1, out float leftToe) ||
                !motion.TryReadFootContact(frame, 2, out float rightHeel) ||
                !motion.TryReadFootContact(frame, 3, out float rightToe))
            {
                return false;
            }

            leftFullyLifted = leftHeel == 0f && leftToe == 0f;
            leftFullyLanded = leftHeel == 1f && leftToe == 1f;
            rightFullyLifted = rightHeel == 0f && rightToe == 0f;
            rightFullyLanded = rightHeel == 1f && rightToe == 1f;
            return true;
        }

[Test]
        public void BuildLoopConstraintJson_UsesFirstFrameAnchorAndEstimatedVelocity()
        {
            const string model = KimodoMotionModelProfiles.ArdyCoreModelName;
            const int frameCount = 4;
            const float frameRate = 20f;
            string[] names = KimodoRigProfileDatabase.GetJointNamesForModel(model);
            var roots = new[]
            {
                new Vector3(2f, 0f, 3f),
                new Vector3(4f, 0f, 5f),
                new Vector3(6f, 0f, 7f),
                new Vector3(8f, 0f, 9f)
            };
            Quaternion firstRootRotation = Quaternion.Euler(20f, 10f, 30f);
            Quaternion tailRootRotation = Quaternion.Euler(-40f, 70f, 80f);
            var rotations = new List<float>(frameCount * names.Length * 4);
            for (int frame = 0; frame < frameCount; frame++)
            {
                for (int joint = 0; joint < names.Length; joint++)
                {
                    Quaternion rotation = joint == 0 && frame == 0 ? firstRootRotation :
                        joint == 0 && frame == frameCount - 1 ? tailRootRotation : Quaternion.identity;
                    rotations.Add(rotation.w);
                    rotations.Add(rotation.x);
                    rotations.Add(-rotation.y);
                    rotations.Add(-rotation.z);
                }
            }
            var footContacts = new byte[frameCount * 4];
            footContacts[0] = 1; // Left foot fully planted before lift.
            footContacts[1] = 1;
            footContacts[12 + 2] = 1; // Right foot fully planted after left lifts.
            footContacts[12 + 3] = 1;

            var motion = new KimodoRawMotionData(
                frameCount,
                names.Length,
                frameRate,
                names,
                KimodoRigProfileDatabase.GetParentIndicesForModel(model),
                roots,
                rotations,
                rootJointIndex: 0,
                footContacts: footContacts);

            JArray constraints = JArray.Parse(KimodoRawMotionConstraintBuilder.BuildLoopConstraintJson(
                motion, model, runtimeTrimStartFrame: 4, targetFrameCount: 8,
                runtimeFrameCount: 16, frameRate: frameRate));

            Assert.That(constraints, Has.Count.EqualTo(2));
            Assert.That((string)constraints[0]["type"], Is.EqualTo("root2d"));
            Assert.That((int)constraints[1]["frame_indices"][0], Is.EqualTo(4));
            Assert.That((float)constraints[1]["root_positions"][0][0], Is.EqualTo(-2f));
            Assert.That((int)constraints[1]["frame_indices"][1], Is.EqualTo(11));
            Assert.That((float)constraints[1]["root_positions"][1][0], Is.EqualTo(-18f));
            Assert.That((int)constraints[0]["frame_indices"][1], Is.EqualTo(15));
            Assert.That((float)constraints[0]["smooth_root_2d"][0][0], Is.EqualTo(6f));
            Assert.That((float)constraints[0]["smooth_root_2d"][0][1], Is.EqualTo(-5f));
            Assert.That((float)constraints[0]["smooth_root_2d"][1][0], Is.EqualTo(-26f));
            Assert.That((float)constraints[0]["smooth_root_2d"][1][1], Is.EqualTo(27f));
            JArray anchorRoot = (JArray)constraints[1]["local_joints_rot"][0][0];
            Quaternion anchorProtocolRotation = KimodoConstraintRotationUtility.AxisAngleVectorToQuaternion(
                new Vector3((float)anchorRoot[0], (float)anchorRoot[1], (float)anchorRoot[2]));
            Quaternion anchorRotation = new Quaternion(
                anchorProtocolRotation.x,
                -anchorProtocolRotation.y,
                -anchorProtocolRotation.z,
                anchorProtocolRotation.w);
            Assert.That(motion.TryReadUnityLocalRotation(0, 0, names.Length, out Quaternion expectedAnchorRotation), Is.True);
            Assert.That(Vector3.Distance(anchorRotation.eulerAngles, expectedAnchorRotation.eulerAngles), Is.LessThan(1e-3f));
            Assert.That(
                ((JArray)constraints[1]["smooth_root_2d"]).Count,
                Is.EqualTo(((JArray)constraints[1]["frame_indices"]).Count));
            JArray terminalRoot = (JArray)constraints[1]["local_joints_rot"][1][0];
            Quaternion protocolRotation = KimodoConstraintRotationUtility.AxisAngleVectorToQuaternion(new Vector3(
                (float)terminalRoot[0], (float)terminalRoot[1], (float)terminalRoot[2]));
            Quaternion terminalRotation = new Quaternion(
                protocolRotation.x, -protocolRotation.y, -protocolRotation.z, protocolRotation.w);
            Quaternion firstYawRotation = Quaternion.Euler(
                0f,
                KimodoMotionMath.ResolvePlanarYawDegrees(firstRootRotation),
                0f);
            Quaternion firstTilt = Quaternion.Inverse(firstYawRotation) * firstRootRotation;
            Vector3 firstTiltEuler = KimodoMotionMath.SignedEulerDegrees(firstTilt);
            Quaternion expectedTerminalRotation = Quaternion.Euler(
                0f,
                KimodoMotionMath.ResolvePlanarYawDegrees(tailRootRotation),
                0f) * Quaternion.Euler(firstTiltEuler.x, 0f, firstTiltEuler.z);
            Assert.That(Quaternion.Angle(terminalRotation, expectedTerminalRotation), Is.LessThan(1e-3f));

            JArray root2D = JArray.Parse(KimodoRawMotionConstraintBuilder.BuildRoot2DConstraintsJson(
                motion, model, new[] { 0, 2 }));
            Assert.That((float)root2D[0]["smooth_root_2d"][1][0], Is.EqualTo(-6f));
        }

private static string BuildAnimationAnalysisSignature(
            TimelineCharacterRecord character,
            TimelineAnimationRecord animation,
            JObject effectiveOptions)
        {
            var signature = new JObject
            {
                // v5 aligns native KMB analysis frames to the fixed 60 FPS command
                // time base before keyframe/contact extraction. Do not reuse
                // v3 caches, whose markers are in the model's native FPS.
                ["contract"] = "animation_analysis_picture_v7_force_line_command_60",
                ["character_ref"] = character?.CharacterRef ?? string.Empty,
                ["rig_type"] = IsHumanoidCharacter(character) ? "humanoid" : "mesh",
                ["animation_id"] = animation?.Id.ToString("D") ?? string.Empty,
                ["start_timeline_frame_60"] = animation?.StartFrame ?? 0,
                ["end_frame_exclusive"] = animation?.EndFrameExclusive ?? 0,
                ["authored_foot_contact_track"] = KimodoTimelineFootContactSampler.HasAuthoredTrack(animation?.Clip),
                ["options"] = CanonicalizeJson(effectiveOptions ?? new JObject())
            };
            return signature.ToString(Formatting.None);
        }

        private static JToken CanonicalizeJson(JToken value)
        {
            if (value is JObject source)
            {
                var result = new JObject();
                foreach (JProperty property in source.Properties().OrderBy(property => property.Name, StringComparer.Ordinal))
                {
                    result[property.Name] = CanonicalizeJson(property.Value);
                }
                return result;
            }
            if (value is JArray array)
            {
                return new JArray(array.Select(CanonicalizeJson));
            }
            return value?.DeepClone() ?? JValue.CreateNull();
        }
